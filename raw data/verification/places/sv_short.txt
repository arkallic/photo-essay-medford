260:     const GAP = 2, PRI = { pin: 0, icon: 1, bubble: 2 };
261:     const live = marks.filter(o => map.hasLayer(o.m));
262:     live.forEach(o => o.m.setLatLng(o.ll));
263:     const box = live.map(o => {
264:       const el = o.m.getElement(), inner = el && el.firstElementChild;
265:       const r = inner ? inner.getBoundingClientRect() : { width: 20, height: 20 };
266:       const p = map.latLngToLayerPoint(o.ll);
267:       return { x: p.x, y: p.y, w: r.width, h: r.height, pri: PRI[o.kind] };
268:     });
269:     for (let it = 0; it < 80; it++) {
270:       let moved = false;
271:       for (let i = 0; i < box.length; i++) for (let j = i + 1; j < box.length; j++) {
272:         const a = box[i], b = box[j];
273:         const dx = b.x - a.x, dy = b.y - a.y;
274:         const px = (a.w + b.w) / 2 + GAP - Math.abs(dx), py = (a.h + b.h) / 2 + GAP - Math.abs(dy);
275:         if (px <= 0 || py <= 0) continue;
276:         if (a.pri === 0 && b.pri === 0) continue;
277:         let sa, sb;   // share of the push each takes
278:         if (a.pri === b.pri) { sa = .5; sb = .5; } else if (a.pri < b.pri) { sa = 0; sb = 1; } else { sa = 1; sb = 0; }
279:         if (px < py) { const s = (dx >= 0 ? 1 : -1) * px; a.x -= s * sa; b.x += s * sb; }
280:         else { const s = (dy >= 0 ? 1 : -1) * py; a.y -= s * sa; b.y += s * sb; }
281:         moved = true;
282:       }
283:       if (!moved) break;
284:     }
